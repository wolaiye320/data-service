```java
@SpringBootTest
@Testcontainers
@Transactional
class UserRepositoryIntegrationTest {

    @Container
    static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:15");

    @Autowired
    private UserRepository repository;

    @Autowired
    private JdbcTemplate jdbcTemplate;

    @DynamicPropertySource
    static void configureProperties(DynamicPropertyRegistry registry) {
        registry.add("spring.datasource.url", postgres::getJdbcUrl);
        registry.add("spring.datasource.username", postgres::getUsername);
        registry.add("spring.datasource.password", postgres::getPassword);
    }

    @Test
    void save_withValidData_shouldPersistCorrectly() {
        // 创建用户
        User user = new User("John", "john@example.com");
        repository.save(user);

        // 验证数据已保存
        User result = repository.findById(user.getId()).orElse(null);
        assertNotNull(result);
        assertEquals("John", result.getName());
        assertEquals("john@example.com", result.getEmail());
    }

    @Test
    void findByEmail_withExistingEmail_shouldReturnUser() {
        // 创建用户
        User user = new User("Jane", "jane@example.com");
        repository.save(user);

        // 查询用户
        Optional<User> result = repository.findByEmail("jane@example.com");
        assertTrue(result.isPresent());
        assertEquals("Jane", result.get().getName());
    }

    @Test
    void findByEmail_withNonExistingEmail_shouldReturnEmpty() {
        Optional<User> result = repository.findByEmail("nonexistent@example.com");
        assertTrue(result.isEmpty());
    }

    @Test
    void update_withValidData_shouldUpdateCorrectly() {
        // 创建用户
        User user = new User("John", "john@example.com");
        repository.save(user);

        // 更新用户
        user.setName("Jane");
        repository.update(user);

        // 验证更新
        User result = repository.findById(user.getId()).orElse(null);
        assertEquals("Jane", result.getName());
        assertEquals("john@example.com", result.getEmail()); // 邮箱不变
    }

    @Test
    void delete_withExistingId_shouldDeleteCorrectly() {
        // 创建用户
        User user = new User("John", "john@example.com");
        repository.save(user);

        // 删除用户
        repository.delete(user.getId());

        // 验证删除
        Optional<User> result = repository.findById(user.getId());
        assertTrue(result.isEmpty());
    }

    @Test
    void findAll_withPagination_shouldReturnCorrectPage() {
        // 创建 15 个用户
        for (int i = 0; i < 15; i++) {
            User user = new User("User" + i, "user" + i + "@example.com");
            repository.save(user);
        }

        // 查询第一页（每页 10 条）
        Page<User> page1 = repository.findAll(PageRequest.of(0, 10));
        assertEquals(10, page1.getContent().size());
        assertEquals(15, page1.getTotalElements());
        assertEquals(2, page1.getTotalPages());

        // 查询第二页
        Page<User> page2 = repository.findAll(PageRequest.of(1, 10));
        assertEquals(5, page2.getContent().size());
    }

    @Test
    void verifyDatabaseSchema() {
        // 验证表包含必要的字段
        jdbcTemplate.execute("SELECT id, name, email, created_at FROM users LIMIT 1");
        // 如果字段不存在，会抛出 SQLException
    }
}
```

```java
@SpringBootTest
@Testcontainers
@Transactional
class UserServiceIntegrationTest {

    @Container
    static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:15");

    @Autowired
    private UserService service;

    @MockBean
    private EmailService emailService;

    @DynamicPropertySource
    static void configureProperties(DynamicPropertyRegistry registry) {
        registry.add("spring.datasource.url", postgres::getJdbcUrl);
        registry.add("spring.datasource.username", postgres::getUsername);
        registry.add("spring.datasource.password", postgres::getPassword);
    }

    @Test
    void createUser_withValidData_shouldPersistAndSendEmail() {
        // 创建用户
        Long userId = service.createUser("John", "john@example.com");

        // 验证用户已保存
        User user = service.getUserById(userId);
        assertNotNull(user);
        assertEquals("John", user.getName());

        // 验证邮件已发送
        verify(emailService, times(1)).sendWelcomeEmail("john@example.com");
    }

    @Test
    void createUser_withDuplicateEmail_shouldThrowException() {
        // 创建第一个用户
        service.createUser("John", "john@example.com");

        // 尝试创建重复邮箱的用户
        assertThrows(DuplicateEmailException.class, () -> {
            service.createUser("Jane", "john@example.com");
        });
    }

    @Test
    void updateUser_withValidData_shouldUpdateCorrectly() {
        // 创建用户
        Long userId = service.createUser("John", "john@example.com");

        // 更新用户
        service.updateUser(userId, "Jane", "jane@example.com");

        // 验证更新
        User user = service.getUserById(userId);
        assertEquals("Jane", user.getName());
        assertEquals("jane@example.com", user.getEmail());
    }

    @Test
    void deleteUser_withExistingId_shouldDeleteCorrectly() {
        // 创建用户
        Long userId = service.createUser("John", "john@example.com");

        // 删除用户
        service.deleteUser(userId);

        // 验证删除
        assertThrows(UserNotFoundException.class, () -> {
            service.getUserById(userId);
        });
    }
}
```

```java
@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
@Testcontainers
class UserApiTest {

    @LocalServerPort
    private int port;

    @Autowired
    private TestRestTemplate restTemplate;

    @Container
    static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:15");

    @MockBean
    private EmailService emailService;

    static void configureProperties(DynamicPropertyRegistry registry) {

        registry.add("spring.datasource.username", postgres::getUsername);
        registry.add("spring.datasource.password", postgres::getPassword);
    }

    @Test
    void createUser_shouldReturnCreatedUser() {
        // 创建用户
        UserRequest request = new UserRequest("John", "john@example.com");
        ResponseEntity<ApiResponse<User>> response = restTemplate.postForEntity(
            "http://localhost:" + port + "/api/v1/users",
            request,
            new ParameterizedTypeReference<>() {}
        );

        // 验证响应
        assertEquals(200, response.getStatusCode().value());
        User user = response.getBody().getData();
        assertEquals("John", user.getName());
        assertEquals("john@example.com", user.getEmail());
        assertNotNull(user.getId());
    }

    @Test
    void getUser_withExistingId_shouldReturnUser() {
        // 创建用户
        UserRequest request = new UserRequest("John", "john@example.com");
        ResponseEntity<ApiResponse<User>> createResponse = restTemplate.postForEntity(
            "http://localhost:" + port + "/api/v1/users",
            request,
            new ParameterizedTypeReference<>() {}
        );
        Long userId = createResponse.getBody().getData().getId();

        // 查询用户
        ResponseEntity<ApiResponse<User>> getResponse = restTemplate.getForEntity(
            "http://localhost:" + port + "/api/v1/users/" + userId,
            new ParameterizedTypeReference<>() {}
        );

        // 验证响应
        assertEquals(200, getResponse.getStatusCode().value());
        User user = getResponse.getBody().getData();
        assertEquals("John", user.getName());
    }

    @Test
    void getUser_withNonExistingId_shouldReturn404() {
        ResponseEntity<ApiResponse<String>> response = restTemplate.getForEntity(
            "http://localhost:" + port + "/api/v1/users/99999",
            new ParameterizedTypeReference<>() {}
        );

        assertEquals(404, response.getStatusCode().value());
    }

    @Test
    void createUser_withInvalidEmail_shouldReturn400() {
        UserRequest request = new UserRequest("John", "invalid-email");
        ResponseEntity<ApiResponse<String>> response = restTemplate.postForEntity(
            "http://localhost:" + port + "/api/v1/users",
            request,
            new ParameterizedTypeReference<>() {}
        );

        assertEquals(400, response.getStatusCode().value());
    }

    @Test
    void listUsers_withPagination_shouldReturnPagedResult() {
        // 创建 15 个用户
        for (int i = 0; i < 15; i++) {
            UserRequest request = new UserRequest("User" + i, "user" + i + "@example.com");
            restTemplate.postForEntity(
                "http://localhost:" + port + "/api/v1/users",
                request,
                new ParameterizedTypeReference<>() {}
            );
        }

        // 查询第一页
        ResponseEntity<ApiResponse<PageResponse<User>>> response = restTemplate.getForEntity(
            "http://localhost:" + port + "/api/v1/users?page=0&size=10",
            new ParameterizedTypeReference<>() {}
        );

        assertEquals(200, response.getStatusCode().value());
        PageResponse<User> page = response.getBody().getData();
        assertEquals(10, page.getContent().size());
        assertEquals(15, page.getTotalElements());
    }
}
```

## Mockito 使用示例

### Mock vs Spy

```java
@SpringBootTest
class MockVsSpyTest {

    // Mock：完全模拟
    @MockBean
    private EmailService emailService;

    // Spy：部分模拟
    @SpyBean
    private UserService userService;

    @Test
    void mockExample() {
        // Mock 所有方法都需要 stub
        when(emailService.sendEmail(anyString(), anyString())).thenReturn(true);

        boolean result = emailService.sendEmail("test@example.com", "Hello");
        assertTrue(result);

        // 验证调用
        verify(emailService).sendEmail("test@example.com", "Hello");
    }

    @Test
    void spyExample() {
        // Spy 只 stub 需要模拟的方法
        doReturn(true).when(userService).isEmailAvailable(anyString());

        // 其他方法调用真实实现
        User user = userService.createUser("John", "john@example.com");
        assertNotNull(user);

        // 验证调用
        verify(userService).isEmailAvailable("john@example.com");
    }
}
```
