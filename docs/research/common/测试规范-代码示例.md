## 1. Repository 集成测试示例

### 1.1 `DS_CONNECTION` Repository 测试

```java
@SpringBootTest
@Testcontainers
@Transactional
class DSConnectionRepositoryIntegrationTest {

    @Container
    static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:15");

    @Autowired
    private DSConnectionRepository repository;

    @Autowired
    private JdbcTemplate jdbcTemplate;

    @DynamicPropertySource
    static void configure(DynamicPropertyRegistry registry) {
        registry.add("spring.datasource.url", postgres::getJdbcUrl);
        registry.add("spring.datasource.username", postgres::getUsername);
        registry.add("spring.datasource.password", postgres::getPassword);
    }

    @Test
    void insert_withValidConnection_shouldPersistSuccessfully() {
        DSConnectionEntity entity = new DSConnectionEntity();
        entity.setConnectionCode("pg_bank");
        entity.setConnectionName("PostgreSQL Bank");
        entity.setDbType("POSTGRESQL");
        entity.setHost("127.0.0.1");
        entity.setPort(5432);
        entity.setUsername("test");
        entity.setPasswordCiphertext("cipher");
        entity.setStatus("ENABLED");

        repository.insert(entity);

        DSConnectionEntity result = repository.findByCode("pg_bank");
        assertThat(result).isNotNull();
        assertThat(result.getDbType()).isEqualTo("POSTGRESQL");
        assertThat(result.getStatus()).isEqualTo("ENABLED");
    }

    @Test
    void findByCode_withExistingCode_shouldReturnMatchedConnection() {
        jdbcTemplate.update("""
            insert into ds_connection
            (connection_code, connection_name, db_type, host, port, username, password_ciphertext, status)
            values (?, ?, ?, ?, ?, ?, ?, ?)
            """,
            "mysql_order", "MySQL Order", "MYSQL", "127.0.0.1", 3306, "root", "cipher", "ENABLED");

        DSConnectionEntity result = repository.findByCode("mysql_order");

        assertThat(result).isNotNull();
        assertThat(result.getConnectionName()).isEqualTo("MySQL Order");
    }

    @Test
    void verifySchema_shouldContainRequiredColumns() {
        jdbcTemplate.execute("""
            select id, connection_code, connection_name, db_type, host, port, username, password_ciphertext, status
            from ds_connection
            limit 1
            """);
    }
}
```

## 2. Service 集成测试示例

### 2.1 数据服务发布校验测试

```java
@SpringBootTest
@Testcontainers
@Transactional
class DataServicePublishServiceIntegrationTest {

    @Container
    static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:15");

    @Autowired
    private DataServicePublishService publishService;

    @Autowired
    private DSRepository dsRepository;

    @DynamicPropertySource
    static void configure(DynamicPropertyRegistry registry) {
        registry.add("spring.datasource.url", postgres::getJdbcUrl);
        registry.add("spring.datasource.username", postgres::getUsername);
        registry.add("spring.datasource.password", postgres::getPassword);
    }

    @Test
    void publish_withCompleteConfiguration_shouldPublishSuccessfully() {
        Long serviceId = prepareCompletedDraftService();

        publishService.publish(serviceId);

        DSEntity result = dsRepository.findById(serviceId);
        assertThat(result.getStatus()).isEqualTo("PUBLISHED");
        assertThat(result.getVersion()).isEqualTo(1);
    }

    @Test
    void publish_withMissingSql_shouldRejectService() {
        Long serviceId = prepareDraftServiceWithoutSql();

        assertThatThrownBy(() -> publishService.publish(serviceId))
            .isInstanceOf(BusinessException.class)
            .hasMessageContaining("SQL");
    }
}
```

## 3. API E2E 测试示例

### 3.1 查询接口测试

```java
@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
@Testcontainers
class DataServiceQueryApiE2ETest {

    @Container
    static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:15");

    @LocalServerPort
    private int port;

    @Autowired
    private TestRestTemplate restTemplate;

    @DynamicPropertySource
    static void configure(DynamicPropertyRegistry registry) {
        registry.add("spring.datasource.url", postgres::getJdbcUrl);
        registry.add("spring.datasource.username", postgres::getUsername);
        registry.add("spring.datasource.password", postgres::getPassword);
    }

    @Test
    void query_withPublishedService_shouldReturnData() {
        QueryRequest request = new QueryRequest(
            "customer_order_query",
            Map.of("customerId", "C1001"),
            List.of()
        );

        ResponseEntity<QueryResponse> response = restTemplate.postForEntity(
            "http://localhost:" + port + "/api/data-services/query",
            request,
            QueryResponse.class
        );

        assertThat(response.getStatusCode().is2xxSuccessful()).isTrue();
        assertThat(response.getBody().getCode()).isEqualTo("SUCCESS");
        assertThat(response.getBody().getData()).isNotEmpty();
    }

    @Test
    void query_withUnknownService_shouldReturnErrorCode() {
        QueryRequest request = new QueryRequest("not_exist_service", Map.of(), List.of());

        ResponseEntity<QueryResponse> response = restTemplate.postForEntity(
            "http://localhost:" + port + "/api/data-services/query",
            request,
            QueryResponse.class
        );

        assertThat(response.getBody().getCode()).isEqualTo("SERVICE_NOT_FOUND");
    }
}
```

## 4. 前端页面级 E2E 示例

### 4.1 Playwright 示例：创建数据服务

```ts
import { test, expect } from "@playwright/test";

test("create data service from page should save successfully", async ({ page }) => {
  await page.goto("/data-services");

  await page.getByRole("button", { name: "新增数据服务" }).click();
  await page.getByLabel("服务编码").fill("customer_order_query");
  await page.getByLabel("服务名称").fill("客户订单查询");
  await page.getByLabel("SQL").fill(
    "select customer_id, order_no, order_amount from customer_order where customer_id = #{customerId}"
  );
  await page.getByRole("button", { name: "保存草稿" }).click();

  await expect(page.getByText("保存成功")).toBeVisible();
  await expect(page.getByText("customer_order_query")).toBeVisible();
});
```
